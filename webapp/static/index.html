Mecânica Clássica

        

          
$$\frac{d}{dt}\left(\frac{\partial L}{\partial \dot{q}_i}\right) - \frac{\partial L}{\partial q_i} = 0, \quad L = T - V$$

          
$$H = \sum p_i \dot{q}_i - L, \quad \dot{q}_i = \frac{\partial H}{\partial p_i}, \quad \dot{p}_i = -\frac{\partial H}{\partial q_i}$$

          
$$V_{\text{ef}}(r) = V(r) + \frac{L^2}{2\mu r^2}, \quad I = \int r_\perp^2 dm$$

        

      


      
      

        
Eletromagnetismo

        

          
$$\nabla \cdot \vec{E} = \frac{\rho}{\varepsilon_0}, \quad \nabla \cdot \vec{B} = 0$$

          
$$\nabla \times \vec{E} = -\frac{\partial \vec{B}}{\partial t}, \quad \nabla \times \vec{B} = \mu_0 \vec{J} + \mu_0 \varepsilon_0 \frac{\partial \vec{E}}{\partial t}$$

          
$$\vec{S} = \frac{1}{\mu_0} (\vec{E} \times \vec{B}), \quad u = \frac{1}{2}\left(\varepsilon_0 E^2 + \frac{B^2}{\mu_0}\right)$$

        

      


      
      

        
Mecânica Quântica

        

          
$$i\hbar \frac{\partial \psi}{\partial t} = \hat{H}\psi, \quad [\hat{x}, \hat{p}] = i\hbar$$

          
$$\hat{a} = \sqrt{\frac{m\omega}{2\hbar}}\left(\hat{x} + \frac{i\hat{p}}{m\omega}\right), \quad [\hat{a}, \hat{a}^\dagger] = 1, \quad E_n = \hbar\omega\left(n + \frac{1}{2}\right)$$

          
$$E_n^{(1)} = \langle \psi_n^{(0)} | \hat{H}' | \psi_n^{(0)} \rangle, \quad [J_i, J_j] = i\hbar \varepsilon_{ijk} J_k$$

$$Z = \sum_i e^{-\beta E_i}, \quad F = -k_B T \ln Z, \quad \langle E \rangle = -\frac{\partial \ln Z}{\partial \beta}$$

          
$$\bar{n}_{\text{FD}} = \frac{1}{e^{\beta(\varepsilon - \mu)} + 1}, \quad \bar{n}_{\text{BE}} = \frac{1}{e^{\beta(\varepsilon - \mu)} - 1}$$
